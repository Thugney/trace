import { useState, useEffect } from 'react';
import { Check, Star, Camera, Play, Pause, Square } from 'lucide-react';
import { useTaskIcon, useAge } from '../context/AgeContext';
import { useProfile } from '../context/ProfileContext';
import { useTimer } from '../hooks/useTimer';
import playSounds from '../utils/sounds';
import confetti from 'canvas-confetti';
import PhotoProof from './PhotoProof';
import { tasksAPI } from '../services/api';
import { useToast } from './Toast';
import { getDeviceId } from '../utils/deviceId';
 
interface SimpleTaskCardProps {
  task: {
    id: number;
    title: string;
    category: string;
    points: number;
    completed: boolean;
    description?: string;
    proof_required?: boolean | number;
    proof_photo_url?: string | null;
    proof_status?: 'pending' | 'approved' | 'rejected' | null;
    estimated_minutes?: number;
  };
  onComplete: (taskId: number) => void;
  disabled?: boolean;
}
 
export default function SimpleTaskCard({ task, onComplete, disabled }: SimpleTaskCardProps) {
  const [isAnimating, setIsAnimating] = useState(false);
  const [showCelebration, setShowCelebration] = useState(false);
  const [showPhotoProof, setShowPhotoProof] = useState(false);
  const { emoji, color } = useTaskIcon(task.category);
  const { activeTimerTaskId, setActiveTimerTaskId, pauseActiveTimer, personalizationConfig } = useProfile();
  const { settings: ageSettings } = useAge();
  const { showToast } = useToast();
 
  const proofRequired = task.proof_required === true || task.proof_required === 1;
  const hasProof = !!task.proof_photo_url;
 
  // Age-adjusted difficulty: adjust estimated minutes for younger children
  const getAgeAdjustedEstimatedMinutes = () => {
    if (!task.estimated_minutes) return task.estimated_minutes;
 
    const isSimpleMode = personalizationConfig?.ageGroup === 'simple' || ageSettings.simplifiedMode;
    if (isSimpleMode && personalizationConfig?.difficulty === 'easy') {
      // For younger children, reduce task duration to make it more achievable
      return Math.max(1, Math.floor(task.estimated_minutes * 0.7)); // 30% shorter
    }
    return task.estimated_minutes;
  };
 
  const ageAdjustedEstimatedMinutes = getAgeAdjustedEstimatedMinutes();
 
  // Timed tasks are those with estimated_minutes > 0 AND not in non-timed categories
  // Non-timed categories: Morning Routine, Checklist, Special/Project (from story AC)
  // Note: category values in database use kebab-case, so we match both formats
  const nonTimedCategoryPatterns = [
    'morning-routine', 'morning_routine', 'morning routine', // Morning Routine
    'checklist', // Checklist
    'special', 'project', 'special-project', 'special_project' // Special/Project
  ];
  const isTimedTask = ageAdjustedEstimatedMinutes && ageAdjustedEstimatedMinutes > 0 && !nonTimedCategoryPatterns.includes(task.category.toLowerCase());
 
  const timer = useTimer({
    taskId: task.id,
    onStart: () => {
      // Pause any other active timer
      if (activeTimerTaskId && activeTimerTaskId !== task.id) {
        pauseActiveTimer();
      }
      setActiveTimerTaskId(task.id);
      playSounds.click();
    },
    onPause: () => {
      playSounds.click();
    },
    onResume: () => {
      playSounds.click();
    },
    onStop: async (totalSeconds) => {
      setActiveTimerTaskId(null);
      playSounds.levelUp();
 
      // Log timer duration to backend (Story 2.3)
      try {
        await tasksAPI.logTimer({
          task_id: task.id,
          timer_type: 'task',
          duration_seconds: totalSeconds,
          total_duration_seconds: totalSeconds
        });
        console.log(`Timer logged for task ${task.id}: ${totalSeconds} seconds`);
      } catch (error) {
        console.error('Failed to log timer duration:', error);
        // Show user-friendly error message
        showToast('Timer logged locally - will sync when back online', 'warning', 5000);
      }
 
      // Show proof upload option if required
      if (proofRequired && !hasProof) {
        setShowPhotoProof(true);
      } else {
        // Auto-complete the task if no proof required
        completeTask();
      }
    },
  });
 
  // Set notification metadata for timer
  useEffect(() => {
    if (isTimedTask) {
      timer.setNotificationMetadata(task.title, ageAdjustedEstimatedMinutes);
    }
  }, [isTimedTask, task.title, ageAdjustedEstimatedMinutes, timer.setNotificationMetadata]);
 
  // Poll for timer sync every 30 seconds only when timer is active
  useEffect(() => {
    if (!isTimedTask || !timer.isRunning) return;
 
    let pollInterval: ReturnType<typeof setInterval>;
    let lastSyncTimestamp: string | undefined;
    let errorCount = 0;
    const MAX_ERRORS_BEFORE_BACKOFF = 3;
 
    const pollTimerSync = async () => {
      try {
        const response = await tasksAPI.getTimerSync(lastSyncTimestamp);
        const { timerStates, timestamp } = response.data;
 
        // Update last sync timestamp for delta updates
        lastSyncTimestamp = timestamp;
        errorCount = 0; // Reset error count on success
 
        // Find timer state for this task
        const taskTimerState = timerStates.find(
          (state) => state.timer_type === 'task' && state.task_id === task.id
        );
 
        if (taskTimerState) {
          // Check if server state is different from local state
          const serverIsRunning = taskTimerState.is_running;
          const serverElapsedSeconds = taskTimerState.elapsed_seconds;
          const serverDeviceId = taskTimerState.device_id;
          const localDeviceId = getDeviceId();
 
          // If timer is running on server but not locally, sync from server
          if (serverIsRunning && !timer.isRunning) {
            // Server says timer is running, but locally it's not
            // This could mean another device started the timer
            console.log(`Timer ${task.id} is running on another device (${serverDeviceId})`);
 
            // Sync timer state from server to this device
            timer.syncFromServer({
              is_running: taskTimerState.is_running,
              is_paused: taskTimerState.is_paused || false,
              elapsed_seconds: taskTimerState.elapsed_seconds,
              start_time: taskTimerState.start_time,
              pause_time: taskTimerState.pause_time,
            });
 
            showToast(`Timer for "${task.title}" synced from another device`, 'info', 3000);
          } else if (serverIsRunning && timer.isRunning) {
            // Both are running - check for drift and device conflict
            const drift = Math.abs(serverElapsedSeconds - timer.elapsedSeconds);
 
            // Check if another device is running the same timer
            if (serverDeviceId && serverDeviceId !== localDeviceId) {
              console.log(`Timer ${task.id} conflict: running on both ${serverDeviceId} and ${localDeviceId}`);
 
              // Pause local timer since server shows it's active elsewhere
              timer.pause();
 
              showToast(`Timer paused - it's active on another device`, 'warning', 5000);
            } else if (drift > 5) {
              // Same device but drift detected - sync from server (server is authoritative)
              console.log(`Timer ${task.id} drift detected: ${drift} seconds - syncing from server`);
 
              timer.syncFromServer({
                is_running: taskTimerState.is_running,
                is_paused: taskTimerState.is_paused || false,
                elapsed_seconds: taskTimerState.elapsed_seconds,
                start_time: taskTimerState.start_time,
                pause_time: taskTimerState.pause_time,
              });
 
              if (drift > 30) { // Significant drift
                showToast(`Timer synchronized (${drift}s drift corrected)`, 'info', 3000);
              }
            }
          }
        }
      } catch (error) {
        console.error('Failed to poll timer sync:', error);
        errorCount++;
 
        // Exponential backoff on repeated errors
        if (errorCount >= MAX_ERRORS_BEFORE_BACKOFF) {
          console.warn(`Too many sync errors (${errorCount}), backing off polling`);
          if (pollInterval) {
            clearInterval(pollInterval);
            // Restart with longer interval
            pollInterval = setInterval(pollTimerSync, 60000); // 1 minute
          }
        }
 
        // Show error toast on first failure
        if (errorCount === 1) {
          showToast('Having trouble syncing timer with server. Will retry.', 'warning', 3000);
        }
      }
    };
 
    // Poll immediately on mount, then every 30 seconds
    pollTimerSync();
    pollInterval = setInterval(() => pollTimerSync(), 30000);
 
    // Listen for visibility change to sync immediately when user returns to tab
    const handleVisibilityChange = () => {
      if (!document.hidden && isTimedTask && timer.isRunning) {
        console.log(`Tab became visible - triggering immediate timer sync for task ${task.id}`);
        pollTimerSync(); // Immediate sync on visibility
      }
    };
 
    document.addEventListener('visibilitychange', handleVisibilityChange);
 
    return () => {
      if (pollInterval) {
        clearInterval(pollInterval);
      }
      document.removeEventListener('visibilitychange', handleVisibilityChange);
    };
  }, [task.id, isTimedTask, timer.isRunning, timer.elapsedSeconds, showToast, task.title]);
 
  const completeTask = () => {
    setIsAnimating(true);
    playSounds.click();
 
    setTimeout(() => {
      onComplete(task.id);
      setShowCelebration(true);
      playSounds.levelUp();
 
      confetti({
        particleCount: 100,
        spread: 70,
        origin: { y: 0.6 },
        colors: ['#ff0000', '#00ff00', '#0000ff', '#ffff00', '#ff00ff']
      });
 
      setTimeout(() => {
        setShowCelebration(false);
        setIsAnimating(false);
      }, 2000);
    }, 300);
  };
 
  const handleTap = async () => {
    if (task.completed || disabled || isAnimating) return;
 
    // For timed tasks, handle timer controls
    if (isTimedTask) {
      if (!timer.isRunning) {
        timer.start();
      } else if (timer.isPaused) {
        timer.start(); // This will resume
      } else {
        timer.pause();
      }
      return;
    }
 
    // For non-timed tasks, handle proof/completion
    if (proofRequired && !hasProof) {
      setShowPhotoProof(true);
      return;
    }
 
    completeTask();
  };
 
  const handleProofUploaded = () => {
    setShowPhotoProof(false);
    completeTask();
  };
 
  // Effect to handle when another timer becomes active
  useEffect(() => {
    if (isTimedTask && timer.isRunning && !timer.isPaused && activeTimerTaskId !== task.id) {
      timer.pause();
    }
  }, [activeTimerTaskId, task.id, timer, isTimedTask]);
 
  // Effect to listen for network status changes from service worker
  useEffect(() => {
    const handleServiceWorkerMessage = (event: MessageEvent) => {
      if (event.data && event.data.type === 'NETWORK_STATUS') {
        const { isOnline } = event.data;
 
        // Show offline indicator when timer is running and we go offline
        if (!isOnline && isTimedTask && timer.isRunning && !timer.isPaused) {
          showToast('Offline - Timer continues locally. Will sync when back online.', 'warning', 5000);
        }
 
        // Show online indicator when we come back online with queued data
        if (isOnline && isTimedTask && timer.isRunning) {
          // Check if there are pending syncs
          const pendingSync = localStorage.getItem('pending_timer_sync');
          if (pendingSync) {
            showToast('Back online - Syncing timer data...', 'info', 3000);
          }
        }
      }
 
      // Handle timer log queued messages
      if (event.data && event.data.type === 'TIMER_LOG_QUEUED') {
        const { durationSeconds } = event.data;
        showToast(`Timer log (${durationSeconds}s) saved locally. Will sync when back online.`, 'info', 5000);
      }
 
      // Handle timer log sync success messages
      if (event.data && event.data.type === 'TIMER_LOG_SYNC_SUCCESS') {
        const { wasQueued } = event.data;
        if (wasQueued) {
          showToast('Timer log successfully synced with server', 'success', 3000);
        }
      }
    };
 
    // Listen for messages from service worker
    if ('serviceWorker' in navigator && navigator.serviceWorker.controller) {
      navigator.serviceWorker.addEventListener('message', handleServiceWorkerMessage);
    }
 
    return () => {
      if ('serviceWorker' in navigator && navigator.serviceWorker.controller) {
        navigator.serviceWorker.removeEventListener('message', handleServiceWorkerMessage);
      }
    };
  }, [isTimedTask, timer.isRunning, timer.isPaused, showToast]);
 
  if (task.completed) {
    return (
      <div className="relative aspect-square rounded-3xl bg-gradient-to-br from-green-100 to-green-200 border-4 border-green-300 flex flex-col items-center justify-center p-4 shadow-lg">
        <div className="absolute -top-2 -right-2 w-12 h-12 bg-green-500 rounded-full flex items-center justify-center shadow-lg animate-bounce">
          <Check className="w-8 h-8 text-white" strokeWidth={3} />
        </div>
        <span className="text-7xl mb-2 opacity-50">{emoji}</span>
        <div className="flex items-center gap-1 text-green-600 font-bold text-xl">
          <Star className="w-6 h-6 fill-current" />
          Done!
        </div>
        {isTimedTask && timer.elapsedSeconds > 0 && (
          <div className="mt-2 text-sm text-green-700 font-medium">
            Time: {timer.formatTime(timer.elapsedSeconds)}
          </div>
        )}
        {hasProof && (
          <>
            {/* Thumbnail preview */}
            <div className="absolute bottom-10 left-1/2 -translate-x-1/2">
              <div className="w-12 h-12 rounded-lg overflow-hidden border-2 border-white shadow-md">
                <img
                  src={task.proof_photo_url?.startsWith('/api') ? task.proof_photo_url : `/api${task.proof_photo_url}`}
                  alt="Proof thumbnail"
                  className="w-full h-full object-cover"
                />
              </div>
            </div>
            {/* Status badge */}
            <div className="absolute bottom-2 left-1/2 -translate-x-1/2">
              <span className={`text-xs flex items-center gap-1 ${task.proof_status === 'pending' ? 'text-yellow-600' : 'text-green-600'}`}>
                <Camera size={12} />
                {task.proof_status === 'pending' ? 'Proof Pending' : 'Photo added'}
              </span>
            </div>
          </>
        )}
      </div>
    );
  }
 
  return (
    <>
      <button
        id={`task-${task.id}`}
        onClick={handleTap}
        disabled={disabled || isAnimating}
        className={`
          relative aspect-square rounded-3xl border-4
          flex flex-col items-center justify-center p-4
          shadow-xl transition-all duration-200
          ${color} border-slate-200
          ${isAnimating ? 'scale-95' : 'hover:scale-105 active:scale-95'}
          ${disabled ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}
          focus:outline-none focus:ring-4 focus:ring-indigo-300
        `}
        style={{ minHeight: '160px', minWidth: '160px' }}
        aria-label={`${task.title} task${isTimedTask ? ' with timer' : ''}${proofRequired ? ' requires photo proof' : ''}`}
      >
        <span className={`text-7xl mb-3 ${isAnimating ? 'animate-bounce' : ''}`}>
          {emoji}
        </span>
 
        <div className="absolute -top-2 -right-2 w-12 h-12 bg-gradient-to-br from-yellow-400 to-orange-500 rounded-full flex items-center justify-center shadow-lg">
          <span className="text-white font-bold text-lg">+{task.points}</span>
        </div>
 
        {proofRequired && (
          <div className="absolute -top-2 -left-2 w-10 h-10 bg-gradient-to-br from-blue-400 to-cyan-500 rounded-full flex items-center justify-center shadow-lg">
            <Camera className="w-5 h-5 text-white" />
          </div>
        )}
 
        <p className="text-lg font-bold text-slate-700 text-center line-clamp-2 px-2">
          {task.title}
        </p>
 
        {/* Visual progress bar for timed tasks */}
        {isTimedTask && ageAdjustedEstimatedMinutes && (
          <div className="w-full mt-2 px-4">
            <div className="h-2 bg-slate-200 rounded-full overflow-hidden">
              <div
                className="h-full bg-blue-500 transition-all duration-1000"
                style={{
                  width: timer.isRunning
                    ? `${Math.min((timer.elapsedSeconds / (ageAdjustedEstimatedMinutes * 60)) * 100, 100)}%`
                    : '0%'
                }}
              />
            </div>
          </div>
        )}
 
        {/* Timer display and controls for timed tasks */}
        {isTimedTask ? (
          <div className="absolute bottom-3 left-1/2 -translate-x-1/2 w-full px-4">
            {timer.isRunning ? (
              <div className="flex flex-col items-center gap-1">
                <div
                  className="text-lg font-bold text-slate-700"
                  role="timer"
                  aria-live="polite"
                  aria-atomic="true"
                  aria-label={`Elapsed time: ${timer.formattedTime}`}
                >
                  {timer.formattedTime}
                </div>
                <div className="flex items-center justify-center gap-2">
                  {timer.isPaused ? (
                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        timer.start();
                      }}
                      className="p-2 bg-green-500 text-white rounded-full hover:bg-green-600 transition-colors focus:outline-none focus:ring-2 focus:ring-green-300 focus:ring-offset-2 min-w-[36px] min-h-[36px]"
                      aria-label={`Resume timer for ${task.title}`}
                      title="Resume timer"
                    >
                      <Play size={16} />
                    </button>
                  ) : (
                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        timer.pause();
                      }}
                      className="p-2 bg-yellow-500 text-white rounded-full hover:bg-yellow-600 transition-colors focus:outline-none focus:ring-2 focus:ring-yellow-300 focus:ring-offset-2 min-w-[36px] min-h-[36px]"
                      aria-label={`Pause timer for ${task.title}`}
                      title="Pause timer"
                    >
                      <Pause size={16} />
                    </button>
                  )}
                  <button
                    onClick={(e) => {
                      e.stopPropagation();
                      timer.stop();
                    }}
                    className="p-2 bg-red-500 text-white rounded-full hover:bg-red-600 transition-colors focus:outline-none focus:ring-2 focus:ring-red-300 focus:ring-offset-2 min-w-[36px] min-h-[36px]"
                    aria-label={`Stop timer for ${task.title}`}
                    title="Stop timer"
                  >
                    <Square size={16} />
                  </button>
                </div>
              </div>
            ) : (
              <div className="flex flex-col items-center">
                <div className="text-sm text-slate-500 mb-1">
                  {ageAdjustedEstimatedMinutes} min
                </div>
                <button
                  onClick={(e) => {
                    e.stopPropagation();
                    timer.start();
                  }}
                  className="px-4 py-2 bg-blue-500 text-white rounded-full text-sm font-medium hover:bg-blue-600 transition-colors flex items-center gap-1 focus:outline-none focus:ring-2 focus:ring-blue-300 focus:ring-offset-2 min-h-[36px]"
                  aria-label={`Start timer for ${task.title} (estimated ${ageAdjustedEstimatedMinutes} minutes)`}
                  title="Start timer"
                >
                  <Play size={14} /> Start Timer
                </button>
              </div>
            )}
          </div>
        ) : (
          <div className="absolute bottom-3 left-1/2 -translate-x-1/2">
            <span className="text-sm text-slate-400 animate-pulse">
              {proofRequired ? '📸 Take photo!' : 'Tap me!'}
            </span>
          </div>
        )}
 
        {showCelebration && (
          <div className="absolute inset-0 bg-green-400/90 rounded-3xl flex flex-col items-center justify-center animate-pulse">
            <span className="text-6xl mb-2">🎉</span>
            <span className="text-2xl font-bold text-white">Great job!</span>
            <span className="text-lg text-white/80">+{task.points} points!</span>
          </div>
        )}
      </button>
 
      {showPhotoProof && (
        <PhotoProof
          taskId={task.id}
          taskTitle={task.title}
          onProofUploaded={handleProofUploaded}
          onCancel={() => setShowPhotoProof(false)}
          existingProofUrl={task.proof_photo_url}
        />
      )}
    </>
  );
}
 
interface SimpleTaskGridProps {
  tasks: SimpleTaskCardProps['task'][];
  onComplete: (taskId: number) => void;
  maxTasks?: number;
}
 
export function SimpleTaskGrid({ tasks, onComplete, maxTasks = 4 }: SimpleTaskGridProps) {
  const incompleteTasks = tasks.filter(t => !t.completed);
  const completedTasks = tasks.filter(t => t.completed);
 
  const visibleTasks = [
    ...incompleteTasks.slice(0, maxTasks),
    ...completedTasks.slice(0, Math.max(0, maxTasks - incompleteTasks.length))
  ];
 
  if (visibleTasks.length === 0) {
    return (
      <div className="flex flex-col items-center justify-center p-8 bg-gradient-to-br from-indigo-50 to-purple-50 rounded-3xl">
        <span className="text-8xl mb-4">🌟</span>
        <h2 className="text-3xl font-bold text-indigo-600 mb-2">All done!</h2>
        <p className="text-xl text-slate-500">Great job today!</p>
      </div>
    );
  }
 
  return (
    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 p-4">
      {visibleTasks.map(task => (
        <SimpleTaskCard
          key={task.id}
          task={task}
          onComplete={onComplete}
        />
      ))}
 
      {incompleteTasks.length > maxTasks && (
        <div className="col-span-2 text-center py-4">
          <span className="text-lg text-slate-500">
            +{incompleteTasks.length - maxTasks} more tasks
          </span>
        </div>
      )}
    </div>
  );
}
