import { createContext, useContext, ReactNode, useMemo } from 'react';
import { useProfile } from './ProfileContext';
 
type AgeGroup = 'young' | 'middle' | 'older' | 'adult';
 
// Theme class constants
const THEME_CLASSES = {
  SIMPLE: 'theme-simple',
  KID_5: 'theme-kid-5',
  KID_7: 'theme-kid-7',
  KID_12: 'theme-kid-12'
} as const;
 
interface AgeSettings {
  ageGroup: AgeGroup;
  simplifiedMode: boolean;
  fontSize: 'normal' | 'large' | 'extra-large';
  iconSize: 'normal' | 'large' | 'extra-large';
  themeClass: string;
  maxTasksPerView: number;
  showTextLabels: boolean;
  useLargeButtons: boolean;
  celebrationIntensity: 'subtle' | 'medium' | 'high';
}
 
interface AgeContextType {
  age: number | null;
  settings: AgeSettings;
  isYoungChild: boolean;
  isMiddleChild: boolean;
  isOlderChild: boolean;
  isAdult: boolean;
}
 
const defaultSettings: AgeSettings = {
  ageGroup: 'adult',
  simplifiedMode: false,
  fontSize: 'normal',
  iconSize: 'normal',
  themeClass: '',
  maxTasksPerView: 20,
  showTextLabels: true,
  useLargeButtons: false,
  celebrationIntensity: 'subtle'
};
 
const AgeContext = createContext<AgeContextType>({
  age: null,
  settings: defaultSettings,
  isYoungChild: false,
  isMiddleChild: false,
  isOlderChild: false,
  isAdult: true
});
 
export function AgeProvider({ children }: { children: ReactNode }) {
  const { currentProfile, personalizationConfig } = useProfile();
 
  const contextValue = useMemo(() => {
    const age = currentProfile?.age || null;
    const profileType = currentProfile?.type || 'parent';
 
    // Determine age group
    let ageGroup: AgeGroup = 'adult';
    if (profileType === 'child' && age !== null) {
      if (age <= 6) {
        ageGroup = 'young';
      } else if (age <= 10) {
        ageGroup = 'middle';
      } else {
        ageGroup = 'older';
      }
    }
 
    // Configure settings based on age group
    let settings: AgeSettings;
 
    switch (ageGroup) {
      case 'young':
        // 5-6 year olds: Picture-based, minimal text, large everything
        settings = {
          ageGroup: 'young',
          simplifiedMode: true,
          fontSize: 'extra-large',
          iconSize: 'extra-large',
          themeClass: THEME_CLASSES.KID_5,
          maxTasksPerView: 4, // Only show 4 tasks at a time
          showTextLabels: false, // Use pictures/emojis instead
          useLargeButtons: true,
          celebrationIntensity: 'high' // Big celebrations!
        };
        break;
 
      case 'middle':
        // 7-10 year olds: Engaging visuals, readable text
        settings = {
          ageGroup: 'middle',
          simplifiedMode: false,
          fontSize: 'large',
          iconSize: 'large',
          themeClass: THEME_CLASSES.KID_7,
          maxTasksPerView: 8,
          showTextLabels: true,
          useLargeButtons: true,
          celebrationIntensity: 'medium'
        };
        break;
 
      case 'older':
        // 11+ year olds: Full features, more mature UI
        settings = {
          ageGroup: 'older',
          simplifiedMode: false,
          fontSize: 'normal',
          iconSize: 'normal',
          themeClass: THEME_CLASSES.KID_12,
          maxTasksPerView: 15,
          showTextLabels: true,
          useLargeButtons: false,
          celebrationIntensity: 'subtle'
        };
        break;
 
      default:
        // Adults: Standard interface
        settings = defaultSettings;
    }
 
    // Apply personalization overrides if they exist
    if (personalizationConfig) {
      // Override simplifiedMode based on UI complexity
      if (personalizationConfig.uiComplexity === 'basic') {
        settings.simplifiedMode = true;
        settings.fontSize = 'extra-large';
        settings.iconSize = 'extra-large';
        settings.useLargeButtons = true;
        settings.showTextLabels = false;
        settings.maxTasksPerView = Math.min(settings.maxTasksPerView, 6);
      } else {
        settings.simplifiedMode = false;
      }
 
      // Override theme class based on personalization theme
      if (personalizationConfig.theme === 'simple') {
        settings.themeClass = THEME_CLASSES.SIMPLE;
      }
      // Note: If theme is 'standard', keep the age-based theme
 
      // Override celebration intensity based on difficulty
      if (personalizationConfig.difficulty === 'easy') {
        settings.celebrationIntensity = 'high';
      }
    }
 
    return {
      age,
      settings,
      isYoungChild: ageGroup === 'young',
      isMiddleChild: ageGroup === 'middle',
      isOlderChild: ageGroup === 'older',
      isAdult: ageGroup === 'adult'
    };
  }, [currentProfile, personalizationConfig]);
 
  return (
    <AgeContext.Provider value={contextValue}>
      {children}
    </AgeContext.Provider>
  );
}
 
export function useAge() {
  return useContext(AgeContext);
}
