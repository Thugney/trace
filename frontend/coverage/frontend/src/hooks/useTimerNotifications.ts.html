import { useEffect, useRef, useCallback } from 'react';
 
interface TimerNotificationOptions {
  taskId: number;
  taskTitle: string;
  estimatedMinutes?: number;
  isRunning: boolean;
  isPaused: boolean;
  elapsedSeconds: number;
  formattedTime: string;
}
 
interface TimerNotificationState {
  notification: Notification | null;
  badgeText: string;
  isSupported: boolean;
  permission: NotificationPermission;
}
 
const TIMER_NOTIFICATION_TAG = 'family-timer-notification';
const BADGE_UPDATE_INTERVAL = 60000; // Update badge every minute
 
/**
 * Custom hook for managing timer notifications and app badges
 * Features:
 * - Shows persistent notification when timer is running in background
 * - Updates app badge with timer status
 * - Quick return to task via notification click
 * - Automatic cleanup when timer stops
 */
export function useTimerNotifications() {
  const notificationRef = useRef<Notification | null>(null);
  const badgeIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const stateRef = useRef<TimerNotificationState>({
    notification: null,
    badgeText: '',
    isSupported: false,
    permission: 'default',
  });
 
  // Check notification support and permission
  useEffect(() => {
    const isSupported = 'Notification' in window && 'serviceWorker' in navigator;
    const permission = Notification.permission;
 
    stateRef.current = {
      ...stateRef.current,
      isSupported,
      permission,
    };
 
    // Clean up any existing notifications on unmount
    return () => {
      cleanupNotifications();
    };
  }, []);
 
  const requestPermission = useCallback(async (): Promise<boolean> => {
    if (!stateRef.current.isSupported) return false;
 
    try {
      const permission = await Notification.requestPermission();
      stateRef.current.permission = permission;
      return permission === 'granted';
    } catch (error) {
      console.error('Failed to request notification permission:', error);
      return false;
    }
  }, []);
 
  const updateAppBadge = useCallback((text: string) => {
    if ('setAppBadge' in navigator) {
      try {
        if (text) {
          (navigator as any).setAppBadge(text);
        } else {
          (navigator as any).clearAppBadge();
        }
        stateRef.current.badgeText = text;
      } catch (error) {
        console.error('Failed to update app badge:', error);
      }
    }
  }, []);
 
  const showTimerNotification = useCallback(async (options: TimerNotificationOptions) => {
    const { taskId, taskTitle, isRunning, isPaused, elapsedSeconds, formattedTime } = options;
 
    if (!stateRef.current.isSupported || stateRef.current.permission !== 'granted') {
      return;
    }
 
    // Clean up existing notification
    if (notificationRef.current) {
      notificationRef.current.close();
      notificationRef.current = null;
    }
 
    // Don't show notification if timer is paused
    if (!isRunning || isPaused) {
      updateAppBadge('');
      return;
    }
 
    try {
      // Calculate time remaining if estimated minutes provided
      let body = `Timer running: ${formattedTime}`;
      if (options.estimatedMinutes) {
        const totalSeconds = options.estimatedMinutes * 60;
        const remainingSeconds = Math.max(0, totalSeconds - elapsedSeconds);
        const remainingMinutes = Math.floor(remainingSeconds / 60);
        const remainingSecs = remainingSeconds % 60;
        body = `${formattedTime} elapsed • ${remainingMinutes}:${remainingSecs.toString().padStart(2, '0')} remaining`;
      }
 
      const notification = new Notification(`⏰ ${taskTitle}`, {
        body,
        icon: '/pwa-192x192.png',
        badge: '/favicon-32x32.png',
        tag: TIMER_NOTIFICATION_TAG,
        requireInteraction: false,
        silent: true,
        data: {
          taskId,
          type: 'timer',
          url: `/tasks?highlight=${taskId}`,
        },
      });
 
      notificationRef.current = notification;
 
      // Update app badge
      updateAppBadge('⏰');
 
      // Set up badge update interval
      if (badgeIntervalRef.current) {
        clearInterval(badgeIntervalRef.current);
      }
 
      badgeIntervalRef.current = setInterval(() => {
        if (notificationRef.current && isRunning && !isPaused) {
          // Update notification body with current time
          const updatedBody = `Timer running: ${formattedTime}`;
          notificationRef.current.body = updatedBody;
 
          // Update app badge
          updateAppBadge('⏰');
        }
      }, BADGE_UPDATE_INTERVAL);
 
      // Handle notification click
      notification.onclick = (event) => {
        event.preventDefault();
 
        // Focus existing window or open new one
        if (window.clients) {
          window.clients.matchAll({ type: 'window', includeUncontrolled: true }).then((clients) => {
            for (const client of clients) {
              if (client.url.includes(window.location.origin) && 'focus' in client) {
                client.focus();
                // Navigate to task
                if ('navigate' in client) {
                  (client as any).navigate(`/tasks?highlight=${taskId}`);
                }
                return;
              }
            }
            // Open new window
            window.open(`/tasks?highlight=${taskId}`, '_blank');
          });
        } else {
          // Fallback for non-PWA context
          window.focus();
          window.location.href = `/tasks?highlight=${taskId}`;
        }
 
        notification.close();
        notificationRef.current = null;
      };
 
      // Handle notification close
      notification.onclose = () => {
        notificationRef.current = null;
      };
 
      console.log('Timer notification shown for task:', taskId);
    } catch (error) {
      console.error('Failed to show timer notification:', error);
    }
  }, [updateAppBadge]);
 
  const updateTimerNotification = useCallback((options: TimerNotificationOptions) => {
    const { isRunning, isPaused, formattedTime } = options;
 
    if (!notificationRef.current) {
      // No active notification, show one if timer is running
      if (isRunning && !isPaused) {
        showTimerNotification(options);
      }
      return;
    }
 
    if (!isRunning || isPaused) {
      // Timer stopped or paused, clean up
      cleanupNotifications();
      return;
    }
 
    // Update existing notification
    try {
      // Update notification body
      let body = `Timer running: ${formattedTime}`;
      if (options.estimatedMinutes) {
        const totalSeconds = options.estimatedMinutes * 60;
        const remainingSeconds = Math.max(0, totalSeconds - options.elapsedSeconds);
        const remainingMinutes = Math.floor(remainingSeconds / 60);
        const remainingSecs = remainingSeconds % 60;
        body = `${formattedTime} elapsed • ${remainingMinutes}:${remainingSecs.toString().padStart(2, '0')} remaining`;
      }
 
      notificationRef.current.body = body;
 
      // Update app badge
      updateAppBadge('⏰');
    } catch (error) {
      console.error('Failed to update timer notification:', error);
    }
  }, [showTimerNotification, updateAppBadge]);
 
  const cleanupNotifications = useCallback(() => {
    if (notificationRef.current) {
      notificationRef.current.close();
      notificationRef.current = null;
    }
 
    if (badgeIntervalRef.current) {
      clearInterval(badgeIntervalRef.current);
      badgeIntervalRef.current = null;
    }
 
    updateAppBadge('');
 
    // Also close any notifications with our tag
    if ('getNotifications' in (navigator as any).serviceWorker) {
      (navigator as any).serviceWorker.getNotifications({ tag: TIMER_NOTIFICATION_TAG })
        .then((notifications: Notification[]) => {
          notifications.forEach(notification => notification.close());
        })
        .catch(console.error);
    }
  }, [updateAppBadge]);
 
  return {
    showTimerNotification,
    updateTimerNotification,
    cleanupNotifications,
    requestPermission,
    isSupported: stateRef.current.isSupported,
    permission: stateRef.current.permission,
  };
}
